        function drawEnemy(enemy) {
            const screenSize = enemy.size;

            if (enemy.type === 'boss') {
                // هیولای بزرگ

                ctx.lineWidth = 4;
                ctx.beginPath();
                ctx.arc(0, screenSize/3, screenSize/2, 0, Math.PI);

        // طراحی گلوله قدرتمند با اثر نور
        function drawBullet(bullet, isEnemy = false) {
            const screenWidth = bullet.width;
            const screenHeight = bullet.height;
            
            // اثر نور و سایه
            if (!isEnemy) {
                const glowGradient = ctx.createRadialGradient(
                    bullet.x, bullet.y, 0,
                    bullet.x, bullet.y, screenWidth * 3
                );
                glowGradient.addColorStop(0, 'rgba(255, 255, 100, 0.8)');
                glowGradient.addColorStop(0.5, 'rgba(255, 200, 50, 0.4)');
                glowGradient.addColorStop(1, 'rgba(255, 150, 0, 0)');
                
                ctx.fillStyle = glowGradient;
                ctx.beginPath();
                ctx.arc(bullet.x, bullet.y, screenWidth * 3, 0, Math.PI * 2);
                ctx.fill();
            }
            
            // بدنه گلوله
            if (isEnemy) {
                ctx.fillStyle = '#ff4444';
                ctx.fillRect(bullet.x - screenWidth/2, bullet.y - screenHeight/2, screenWidth, screenHeight);
                
                ctx.fillStyle = '#ff0000';
                ctx.fillRect(bullet.x - screenWidth/2, bullet.y - screenHeight/2, screenWidth, screenHeight/2);
            } else {
                const bulletGradient = ctx.createLinearGradient(
                    bullet.x - screenWidth/2, bullet.y,
                    bullet.x + screenWidth/2, bullet.y
                );
                bulletGradient.addColorStop(0, '#ffff00');
                bulletGradient.addColorStop(0.5, '#ffaa00');
                bulletGradient.addColorStop(1, '#ffff00');
                
                ctx.fillStyle = bulletGradient;
                ctx.fillRect(bullet.x - screenWidth/2, bullet.y - screenHeight/2, screenWidth, screenHeight);
                
                ctx.fillStyle = '#ff9900';
                ctx.fillRect(bullet.x - screenWidth/2, bullet.y - screenHeight/2, screenWidth, screenHeight/3);
            }
            
            // دنباله گلوله
            if (!isEnemy) {
                ctx.fillStyle = 'rgba(255, 200, 50, 0.6)';
                ctx.fillRect(bullet.x - screenWidth/4, bullet.y + screenHeight/2, screenWidth/2, screenHeight * 2);
            }
        }
        
        // طراحی انفجار پیشرفته
        function drawExplosion(explosion) {
            const screenSize = explosion.size * (explosion.life / explosion.maxLife);
            
            const gradient = ctx.createRadialGradient(
                explosion.x, explosion.y, 0,
                explosion.x, explosion.y, screenSize
            );
            gradient.addColorStop(0, '#ffff00');
            gradient.addColorStop(0.2, '#ffaa00');
            gradient.addColorStop(0.5, '#ff5500');
            gradient.addColorStop(0.8, '#ff0000');
            gradient.addColorStop(1, 'rgba(255, 0, 0, 0)');
            
            ctx.fillStyle = gradient;
            ctx.beginPath();
            ctx.arc(explosion.x, explosion.y, screenSize, 0, Math.PI * 2);
            ctx.fill();
            
            // موج انفجار
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.8)';
            ctx.lineWidth = 4;
            ctx.beginPath();
            ctx.arc(explosion.x, explosion.y, screenSize * 1.3, 0, Math.PI * 2);
            ctx.stroke();
            
            // ذرات انفجار
            for (let i = 0; i < 5; i++) {
                const angle = (explosion.life / explosion.maxLife) * Math.PI * 2 + (i * Math.PI * 0.4);
                const particleSize = screenSize * 0.3;
                const px = explosion.x + Math.cos(angle) * screenSize * 0.8;
                const py = explosion.y + Math.sin(angle) * screenSize * 0.8;
                
                ctx.fillStyle = i % 2 === 0 ? '#ffff00' : '#ffaa00';
                ctx.beginPath();
                ctx.arc(px, py, particleSize, 0, Math.PI * 2);
                ctx.fill();
            }
        }

            ctx.beginPath();
            ctx.arc(coin.x, coin.y, coin.size * 0.7, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.fillStyle = '#ffff00';
            ctx.beginPath();
            ctx.arc(coin.x, coin.y, coin.size * 0.5, 0, Math.PI * 2);
            ctx.fill();
            
            ctx.save();
            ctx.translate(coin.x, coin.y);
            ctx.rotate(coin.rotation);
            ctx.fillStyle = '#ff6600';
            ctx.fillRect(-coin.size * 0.3, -coin.size * 0.1, coin.size * 0.6, coin.size * 0.2);
            ctx.restore();
        }
        
        // طراحی پس‌زمینه متحرک
        function drawBackground() {
            if (backgroundImage && backgroundImage.complete) {
                // رسم تصویر پس‌زمینه
                const imgHeight = canvas.height;
                const imgWidth = canvas.width;
                
                ctx.drawImage(backgroundImage, 0, backgroundY, imgWidth, imgHeight);
                ctx.drawImage(backgroundImage, 0, backgroundY - imgHeight, imgWidth, imgHeight);
            } else {
                // پس‌زمینه تاریک در صورت عدم بارگذاری تصویر
                ctx.fillStyle = '#0a1525';
                ctx.fillRect(0, 0, canvas.width, canvas.height);
            }
        }
        
        // به روزرسانی پس‌زمینه متحرک
        function updateBackground() {
            backgroundY += backgroundSpeed;
            if (backgroundY >= canvas.height) {
                backgroundY = 0;
            }
        }
        
        // ایجاد دشمن جدید بر اساس مرحله
        function spawnEnemy() {
            const rand = Math.random();
            let enemyType = 'fighter';
            let size = 40;
            let health = 15;
            let speed = 2 + Math.random() * 2;
            
            // انتخاب نوع دشمن بر اساس مرحله
            if (currentLevel >= 5 && rand < 0.15) {
                enemyType = 'meteor';
                size = 50;
                health = 10;
                speed = 4 + Math.random() * 3;
            } else if (currentLevel >= 8 && rand < 0.25) {
                enemyType = 'tank';
                size = 70;
                health = 40;
                speed = 1 + Math.random();
            } else if (currentLevel >= 3) {
                size = 40 + Math.random() * 30;
                health = 15 + currentLevel * 2;
            }
            
            enemies.push({
                x: Math.random() * canvas.width,
                y: -50,
                size: size,
                speed: speed,
                health: health,
                maxHealth: health,
                lastShot: 0,
                type: enemyType
            });
        }
        
        // ایجاد هیولای بزرگ
        function spawnBoss() {
            enemies.push({
                x: canvas.width / 2,
                y: -100,
                size: 120,
                speed: 1,
                health: 200,
                maxHealth: 200,
                lastShot: 0,
                type: 'boss'
            });
        }
        
        // ایجاد آدم‌خوار ترسناک
        function spawnZombie() {
            const count = Math.floor(currentLevel / 3) + 1;
            for (let i = 0; i < count; i++) {
                enemies.push({
                    x: Math.random() * canvas.width,
                    y: -50,
                    size: 50,
                    speed: 2 + Math.random() * 2,
                    health: 30,
                    maxHealth: 30,
                    lastShot: 0,
                    type: 'zombie'
                });
            }
        }
        
        // شلیک گلوله قدرتمند بازیکن
        function shoot() {
            const currentTime = Date.now();
            if (currentTime - lastShot > shootDelay) {
                bullets.push({
                    x: playerX,
                    y: playerY - 30,
                    width: 15,
                    height: 35,
                    speed: 15,
                    damage: 20
                });
                
                bullets.push({
                    x: playerX - 25,
                    y: playerY - 25,
                    width: 12,
                    height: 30,
                    speed: 15,
                    damage: 15
                });
                
                bullets.push({
                    x: playerX + 25,
                    y: playerY - 25,
                    width: 12,
                    height: 30,
                    speed: 15,
                    damage: 15
                });
                
                lastShot = currentTime;
            }
        }
        
        // شلیک دشمنان - هر 10 ثانیه 2 تیر
        function enemyShoot() {
            const currentTime = Date.now();
            
            for (const enemy of enemies) {
                if (currentTime - enemy.lastShot > 10000) {
                    // شلیک دو تیر
                    for (let i = 0; i < 2; i++) {
                        enemyBullets.push({
                            x: enemy.x + (i === 0 ? -10 : 10),
                            y: enemy.y + enemy.size/2,
                            width: 8,
                            height: 20,
                            speed: 5,
                            damage: 1
                        });
                    }
                    enemy.lastShot = currentTime;
                }
            }
        }
        
        // ایجاد انفجار
        function createExplosion(x, y, size) {
            explosions.push({
                x: x,
                y: y,
                size: size,
                life: 40,
                maxLife: 40
            });
        }
        
        // ایجاد سکه
        function createCoins(x, y, count) {
            for (let i = 0; i < count; i++) {
                coinsParticles.push({
                    x: x,
                    y: y,
                    size: 12 + Math.random() * 8,
                    speedX: (Math.random() - 0.5) * 8,
                    speedY: 2 + Math.random() * 4,
                    rotation: Math.random() * Math.PI * 2,
                    rotationSpeed: (Math.random() - 0.5) * 0.2,
                    life: 120
                });
            }
        }
        
        // به روزرسانی امتیاز
        function updateScore() {
            scoreElement.textContent = score;
        }
        
        // به روزرسانی سکه
        function updateCoins() {
            coinsElement.textContent = coins;
        }
        
        // به روزرسانی مرحله
        function updateLevel() {
            levelElement.textContent = currentLevel;
        }
        
        // به روزرسانی سلامت
        function updateHealth() {
            const healthPercent = (playerHealth / maxPlayerHealth) * 100;
            healthFill.style.width = `${healthPercent}%`;
        }
        
        // تشخیص برخورد
        function checkCollision(obj1, obj2, radius1, radius2) {
            const dx = obj1.x - obj2.x;
            const dy = obj1.y - obj2.y;
            const distance = Math.sqrt(dx*dx + dy*dy);
            return distance < (radius1 + radius2);
        }
        
        // رفتن به مرحله بعدی
        function nextLevel() {
            currentLevel++;
            enemiesDestroyedInLevel = 0;
            enemiesNeededForLevel = 10 + (currentLevel * 2);
            updateLevel();
            
            // پخش صدای انتقال مرحله
            try {
                levelTransitionSound.play().catch(e => console.log('صدا پخش نشد'));
            } catch(e) {}
            
            // تغییر پس‌زمینه
            currentBackgroundIndex = ((currentLevel - 1) % 100) + 1;
            loadBackground();
            
            // بهبود سلامت بازیکن
            playerHealth = Math.min(maxPlayerHealth, playerHealth + 50);
            updateHealth();
        }
        
        // بازی تمام شد
        function gameOver() {
            gameActive = false;
            finalScoreElement.textContent = score;
            finalCoinsElement.textContent = coins;
            finalLevelElement.textContent = currentLevel;
            gameOverScreen.classList.remove('hidden');
            
            // توقف صداها
            try {
                jetSound.pause();
                backgroundMusic.pause();
            } catch(e) {}
        }
        
        // حلقه اصلی بازی
        function gameLoop() {
            if (!gameActive) return;
            
            drawBackground();
            updateBackground();
            
            // انیمیشن چرخش جنگنده
            playerRoll = (playerX - canvas.width/2) / canvas.width * 25;
            
            // شلیک خودکار
            shoot();
            enemyShoot();
            
            // بررسی زمان برای ایجاد هیولا و آدم‌خوار
            const currentTime = Date.now();
            if (currentTime - lastBossSpawn > bossSpawnInterval && currentLevel >= 10) {
                spawnBoss();
                lastBossSpawn = currentTime;
            }
            
            if (currentTime - lastZombieSpawn > zombieSpawnInterval && currentLevel >= 5) {
                spawnZombie();
                lastZombieSpawn = currentTime;
            }
            
            // رسم بازیکن
            drawPlayer();
            
            // به روزرسانی و رسم گلوله‌های قدرتمند بازیکن
            for (let i = bullets.length - 1; i >= 0; i--) {
                const bullet = bullets[i];
                bullet.y -= bullet.speed;
                
                if (bullet.y < -50) {
                    bullets.splice(i, 1);
                    continue;
                }
                
                drawBullet(bullet);
                
                // بررسی برخورد گلوله با دشمن
                for (let j = enemies.length - 1; j >= 0; j--) {
                    const enemy = enemies[j];
                    if (checkCollision(bullet, enemy, 10, enemy.size/2)) {
                        enemy.health -= bullet.damage;
                        
                        createExplosion(bullet.x, bullet.y, 20);
                        
                        if (enemy.health <= 0) {
                            createExplosion(enemy.x, enemy.y, enemy.size);
                            const coinCount = enemy.type === 'boss' ? 20 : (enemy.type === 'zombie' ? 8 : 3 + Math.floor(Math.random() * 4));
                            createCoins(enemy.x, enemy.y, coinCount);
                            enemies.splice(j, 1);
                            
                            const scoreGain = enemy.type === 'boss' ? 500 : (enemy.type === 'zombie' ? 100 : 25);
                            score += scoreGain;
                            updateScore();
                            
                            enemiesDestroyedInLevel++;
                            if (enemiesDestroyedInLevel >= enemiesNeededForLevel) {
                                nextLevel();
                            }
                        }
                        
                        bullets.splice(i, 1);
                        break;
                    }
                }
            }
            
            // به روزرسانی و رسم گلوله‌های دشمن
            for (let i = enemyBullets.length - 1; i >= 0; i--) {
                const bullet = enemyBullets[i];
                bullet.y += bullet.speed;
                
                if (bullet.y > canvas.height + 50) {
                    enemyBullets.splice(i, 1);
                    continue;
                }
                
                drawBullet(bullet, true);
                
                // بررسی برخورد گلوله دشمن با بازیکن
                if (checkCollision(bullet, {x: playerX, y: playerY}, 8, 30)) {
                    createExplosion(bullet.x, bullet.y, 30);
                    enemyBullets.splice(i, 1);
                    playerHealth -= bullet.damage;
                    updateHealth();
                    
                    if (playerHealth <= 0) {
                        createExplosion(playerX, playerY, 100);
                        gameOver();
                        return;
                    }
                }
            }
            
            // ایجاد دشمنان جدید
            if (Math.random() < 0.02) {
                spawnEnemy();
            }
            
            // به روزرسانی و رسم دشمنان
            for (let i = enemies.length - 1; i >= 0; i--) {
                const enemy = enemies[i];
                enemy.y += enemy.speed;
                
                if (enemy.y > canvas.height + 100) {
                    enemies.splice(i, 1);
                    continue;
                }
                
                drawEnemy(enemy);
                
                // بررسی برخورد دشمن با بازیکن
                if (checkCollision(enemy, {x: playerX, y: playerY}, enemy.size/2, 30)) {
                    createExplosion(enemy.x, enemy.y, enemy.size);
                    createExplosion(playerX, playerY, 80);
                    enemies.splice(i, 1);
                    playerHealth -= 25;
                    updateHealth();
                    
                    if (playerHealth <= 0) {
                        gameOver();
                        return;
                    }
                }
            }
            
            // به روزرسانی و رسم انفجارها
            for (let i = explosions.length - 1; i >= 0; i--) {
                const explosion = explosions[i];
                explosion.life--;
                
                if (explosion.life <= 0) {
                    explosions.splice(i, 1);
                    continue;
                }
                
                drawExplosion(explosion);
            }
            
            // به روزرسانی و رسم سکه‌ها
            for (let i = coinsParticles.length - 1; i >= 0; i--) {
                const coin = coinsParticles[i];
                coin.x += coin.speedX;
                coin.y += coin.speedY;
                coin.rotation += coin.rotationSpeed;
                coin.life--;
                
                if (coin.life <= 0 || coin.y > canvas.height + 50) {
                    coinsParticles.splice(i, 1);
                    continue;
                }
                
                // بررسی جمع‌آوری سکه
                if (checkCollision(coin, {x: playerX, y: playerY}, coin.size, 30)) {
                    coins++;
                    updateCoins();
                    coinsParticles.splice(i, 1);
                    continue;
                }
                
                drawCoin(coin);
            }
            
            // ادامه حلقه بازی
            requestAnimationFrame(gameLoop);
        }
        
        // مدیریت کنترل لمسی
        canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            const touch = e.touches[0];
            dragStartX = touch.clientX;
            dragStartY = touch.clientY;
            isDragging = true;
        });
        
        canvas.addEventListener('touchmove', (e) => {
            if (!isDragging) return;
            e.preventDefault();
            const touch = e.touches[0];
            
            const deltaX = touch.clientX - dragStartX;
            const deltaY = touch.clientY - dragStartY;
            
            playerX += deltaX;
            playerY += deltaY;
            
            playerX = Math.max(40, Math.min(canvas.width - 40, playerX));
            playerY = Math.max(40, Math.min(canvas.height - 40, playerY));
            
            dragStartX = touch.clientX;
            dragStartY = touch.clientY;
        });
        
        canvas.addEventListener('touchend', (e) => {
            e.preventDefault();
            isDragging = false;
        });
        
        // شروع بازی
        startBtn.addEventListener('click', () => {
            startScreen.classList.add('hidden');
            gameActive = true;
            initGame();
            gameLoop();
        });
        
        // شروع مجدد بازی
        restartBtn.addEventListener('click', () => {
            gameOverScreen.classList.add('hidden');
            gameActive = true;
            initGame();
            gameLoop();
        });
        
        // جلوگیری از اسکرول با لمس
        document.body.addEventListener('touchmove', (e) => {
            if (gameActive) {
                e.preventDefault();
            }
        }, { passive: false });
    </script>
